5-Minute Integration Recipes

Select your stack to view ready-to-run drop-in code.

// TypeScript / Node.js 18+ import { ZTDSClient } from '@privacyscrubber/sdk'; import OpenAI from 'openai'; const openai = new OpenAI(); const ztds = new ZTDSClient({ entities: ['NAMES', 'EMAILS', 'PHONE_NUMBERS', 'FINANCIAL_IDS', 'SECRETS'], reversible: true }); async function askAI(userPrompt: string) { // 1. Strip sensitive data locally in client volatile memory const { safePrompt, tokenMap } = ztds.sanitize(userPrompt); // 2. Send sanitized prompt to OpenAI (0 bytes cleartext PII exposed) const completion = await openai.chat.completions.create({ model: 'gpt-4o', messages: [{ role: 'user', content: safePrompt }] }); // 3. Re-hydrate AI response with real values locally return ztds.restore(completion.choices[0].message.content, tokenMap); }
# Python 3.9+ Bridge from ztds import ZTDSClient import openai client = ZTDSClient() def safe_ai_call(prompt: str) -> str: # Sanitize locally result = client.sanitize(prompt) # Call OpenAI response = openai.chat.completions.create( model="gpt-4o", messages=[{"role": "user", "content": result.safe_prompt}] ) # Restore locally return client.restore(response.choices[0].message.content, result.token_map)
// LangChain Expression Language (LCEL) Middleware import { RunnableLambda } from '@langchain/core/runnables'; import { ZTDSClient } from '@privacyscrubber/sdk'; const ztds = new ZTDSClient(); const sanitizeStep = new RunnableLambda({ func: (input: { prompt: string }) => { const { safePrompt, tokenMap } = ztds.sanitize(input.prompt); return { safePrompt, tokenMap }; } });
