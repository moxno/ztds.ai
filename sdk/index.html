LangChain BaseCallbackHandler Middleware Intercepts prompts and completions transparently across any LCEL chain, ChatOpenAI, ChatAnthropic, or LangGraph agent without modifying existing pipeline logic. 
// ztds-langchain-callback.ts
import { BaseCallbackHandler } from "@langchain/core/callbacks/base";
import { LLMResult } from "@langchain/core/outputs";
import { ZTDSEngine } from "@privacyscrubber/sdk";
import { ChatOpenAI } from "@langchain/openai";

export class ZTDSCallbackHandler extends BaseCallbackHandler {
  name = "ZTDSCallbackHandler";
  private engine: ZTDSEngine;
  private sessionMaps: Map<string, Record<string, string>> = new Map();

  constructor(profile: "general" | "healthcare" | "fintech" | "legal" = "general") {
    super();
    this.engine = new ZTDSEngine({ profile });
  }

  // 1. Intercept and sanitize prompts before network dispatch
  async handleLLMStart(llm: { name: string }, prompts: string[], runId: string) {
    for (let i = 0; i < prompts.length; i++) {
      const { sanitizedPrompt, sessionMap } = this.engine.sanitize(prompts[i]);
      prompts[i] = sanitizedPrompt; // In-place payload replacement
      this.sessionMaps.set(runId, sessionMap);
    }
  }

  // 2. Losslessly restore cleartext strictly inside local volatile RAM
  async handleLLMEnd(output: LLMResult, runId: string) {
    const sessionMap = this.sessionMaps.get(runId);
    if (!sessionMap) return;

    for (const generations of output.generations) {
      for (const gen of generations) {
        gen.text = this.engine.reveal(gen.text, sessionMap);
      }
    }
    this.sessionMaps.delete(runId); // Wipe sessionMap from RAM immediately
  }
}

// Drop-in usage:
const model = new ChatOpenAI({
  modelName: "gpt-4o",
  callbacks: [new ZTDSCallbackHandler("healthcare")]
});
LlamaIndex IngestionPipeline TransformComponent Immunizes vector stores (Pinecone, Qdrant, Chroma, Milvus) against permanent GDPR Article 17 vector poisoning by sanitizing documents before embedding creation. 
# ztds_llamaindex_transform.py
from typing import List, Optional
from llama_index.core.schema import BaseNode, TransformComponent
from llama_index.core.ingestion import IngestionPipeline
from llama_index.core.node_parser import SentenceSplitter
from privacyscrubber import ZTDSEngine

class ZTDSTransformation(TransformComponent):
    """Pre-indexing ingestion sanitizer preventing raw PII vector storage."""
    profile: str = "fintech"

    def __init__(self, profile: str = "fintech"):
        super().__init__()
        self.profile = profile
        self._engine = ZTDSEngine(profile=self.profile)

    def __call__(self, nodes: List[BaseNode], **kwargs) -> List[BaseNode]:
        for node in nodes:
            # 1. Deterministic surrogate replacement ([IBAN_1], [SSN_1])
            sanitized_text, token_map = self._engine.sanitize(node.get_content())
            node.set_content(sanitized_text)

            # 2. Add cryptographically verifiable audit tags
            node.metadata["ztds_sanitized"] = True
            node.metadata["ztds_tokens_masked"] = len(token_map)
        return nodes

# Production RAG Ingestion Pipeline
pipeline = IngestionPipeline(
    transformations=[
        SentenceSplitter(chunk_size=512, chunk_overlap=32),
        ZTDSTransformation(profile="fintech") # Vector DB receives 0 bytes cleartext
    ]
)
nodes = pipeline.run(documents=documents)

import { ZTDSEngine } from "@privacyscrubber/sdk";
import OpenAI from "openai";

const openai = new OpenAI();
const ztds = new ZTDSEngine({
  profile: "healthcare", // "general" | "healthcare" | "fintech" | "legal" | "devops"
  maxLatencyMs: 2.0      // Invariant 3: Hard abort if tokenization takes >2ms
});

async function callSecureAI(userPrompt: string): Promise<string> {
  // 1. Strip sensitive entities in local volatile RAM (<0.8ms, 0.00 bytes egress)
  const { sanitizedPrompt, sessionMap, latencyMs } = ztds.sanitize(userPrompt);

  // 2. Transmit sanitized prompt to external LLM provider
  const completion = await openai.chat.completions.create({
    model: "gpt-4o",
    messages: [{ role: "user", content: sanitizedPrompt }]
  });

  // 3. Losslessly restore cleartext strictly inside local memory
  const rawResponse = completion.choices[0].message.content || "";
  return ztds.reveal(rawResponse, sessionMap);
}
